function searchable(raw){
  let text='',map=[],previous='';
  for(let i=0;i<raw.length;i++){
    const current=raw[i];

}

function reviewIssues(row){
  const current=(row.review?.issues||[]).filter(issue=>Object.entries(issue.observed||{}).every(([key,value])=>(row[key]??null)===(value??null)));
  const seen=new Set(current.map(issue=>issue.code));
  const local=(code,field,ru,kk,hintRu,hintKk)=>{if(!seen.has(code))current.push({code,field,severity:'warning',message:{ru,kk},suggestion:{ru:hintRu,kk:hintKk},candidates:[]});};
  if(!row.house)local('missing_house','house','Не найден номер дома.','Үй нөмірі табылмады.','Укажите номер дома по исходной строке.','Үй нөмірін бастапқы жол бойынша енгізіңіз.');
  if(!row.street_name&&!row.living_area)local('missing_street','street_name','Не найдена улица или жилмассив.','Көше не тұрғын алап табылмады.','Уточните название улицы или жилмассива.','Көше не тұрғын алап атауын нақтылаңыз.');
  if(!row.city)local('missing_city','city','Не найден город.','Қала табылмады.','Уточните город: это поможет проверить район и область.','Қаланы нақтылаңыз: бұл аудан мен облысты тексеруге көмектеседі.');
  if(row.postal_code&&!/^(?:[0-9]{6}|[a-z][0-9]{2}[a-z][0-9][a-z][0-9])$/i.test(row.postal_code))local('invalid_postal','postal_code','Необычный формат почтового индекса.','Пошта индексінің пішімі ерекше.','Проверьте индекс; не дополняйте его случайными цифрами.','Индексті тексеріңіз; кездейсоқ сан қоспаңыз.');
  if(confidence(row)<threshold&&!current.some(issue=>issue.severity==='warning'))local('low_confidence',null,'Уверенность ниже выбранного порога.','Сенімділік таңдалған шектен төмен.','Сверьте все поля и подтвердите результат вручную.','Барлық өрісті салыстырып, нәтижені қолмен растаңыз.');

}

@contextmanager
def database():
    DB_PATH.parent.mkdir(parents=True, exist_ok=True)
    db = sqlite3.connect(DB_PATH, timeout=10)
    try:
        with db:
            db.execute('CREATE TABLE IF NOT EXISTS users (id INTEGER PRIMARY KEY, name TEXT NOT NULL, email TEXT UNIQUE NOT NULL, salt TEXT NOT NULL, password_hash TEXT NOT NULL)')
            db.execute('CREATE TABLE IF NOT EXISTS sessions (token_hash TEXT PRIMARY KEY, user_id INTEGER NOT NULL, expires REAL NOT NULL)')
            yield db
    finally:
        db.close()

def check_origin(request):
    origin = request.headers.get('origin')
    expected = f'{request.url.scheme}://{request.url.netloc}'

def email_key(value):
    value = value.strip().lower()
    if not re.fullmatch(r'[^\s@]+@[^\s@]+\.[^\s@]+', value):
        raise HTTPException(400, 'invalid_email')
    return value


def password_hash(password, salt):
    return hashlib.pbkdf2_hmac('sha256', password.encode(), bytes.fromhex(salt), ITERATIONS).hex()


def rate_limit(request):
    key = request.client.host if request.client else 'local'
    now = time.monotonic()
    with _lock:
        recent = [t for t in _attempts[key] if t > now - 900]
        if len(recent) >= 20:
            raise HTTPException(429, 'too_many_attempts')
        _attempts[key] = recent + [now]


def start_session(db, user, request, response):
    old = request.cookies.get(COOKIE)
    if old:
        db.execute('DELETE FROM sessions WHERE token_hash=?', (hashlib.sha256(old.encode()).hexdigest(),))
    token = secrets.token_urlsafe(32)
    db.execute('DELETE FROM sessions WHERE expires < ?', (time.time(),))
    db.execute('INSERT INTO sessions VALUES (?, ?, ?)', (hashlib.sha256(token.encode()).hexdigest(), user[0], time.time() + SESSION_TTL))
    response.set_cookie(COOKIE, token, max_age=SESSION_TTL, httponly=True, secure=request.url.scheme == 'https', samesite='strict', path='/')
    response.headers['Cache-Control'] = 'no-store'
    return {'user': {'id': user[0], 'name': user[1], 'email': user[2]}}


@router.post('/register', status_code=201)

    rate_limit(request)
    email = email_key(data.email)
    name = data.name.strip()
    if not name:

import csv
import os
import re
from app.normalization import CITY_ALIASES, PROVINCE_ALIASES

PATH = os.path.join(os.path.dirname(__file__), "..", "reference", "kato_ru.csv")

KZ_LETTERS = re.compile(r"[әіңғүұқөһӘІҢҒҮҰҚӨҺ]")
DOTTED = re.compile(
    r"(?<!\w)(?:г\.а\.|г\.|гор\.|с\.о\.|с\.|п\.|пос\.|а\.о\.|а\.|ст\.|обл\.)\s*", re.I)
WORDS = re.compile(
    r"\b(?:город|область|район|поселок|посёлок|село|р-н|қаласы|қала|облысы|облыс|ауданы|аудан|ауыл|ауылы|кент)\b", re.I)


def strip_markers(t: str) -> str:
    t = DOTTED.sub(" ", t)
    t = WORDS.sub(" ", t)
    return re.sub(r"\s+", " ", t).strip(" .,-")


def norm(t: str) -> str:
    t = t.lower().replace("ё", "е")
    t = re.sub(r"[^\w\s-]", " ", t)
    t = re.sub(r"\s+", " ", t).strip()
    return CITY_ALIASES.get(t, t)


def pick_ru(seg: str) -> str:
    """'город Алматы/Алматы қаласы' -> 'город Алматы' (русская половина)."""
    parts = [p.strip() for p in seg.split("/")]
    for p in parts:
        if p and not KZ_LETTERS.search(p):
            return p
    return parts[0]


ROWS = {}
with open(PATH, encoding="utf-8-sig", newline="") as f:
    for r in csv.DictReader(f, delimiter=";"):
        ROWS[int(r["id"])] = (int(r["parent_id"]), r["name"].strip())


def top_of(i):
    while ROWS[i][0] in ROWS:
        i = ROWS[i][0]
    return i


def depth_of(i):
    d = 0
    while ROWS[i][0] in ROWS:
        i = ROWS[i][0]
        d += 1
    return d


PROV, DIST, LOC = {}, {}, {}
for i, (parent, name) in ROWS.items():
    key = norm(strip_markers(name))
    if not key:
        continue
    d = depth_of(i)
    if d == 0:
        PROV.setdefault(key, []).append(i)
        if not re.search(r"област", name, re.I):
            LOC.setdefault(key, []).append(i)
    elif re.search(r"с\.о|а\.о", name, re.I):
        continue
    elif re.search(r"район|р\.а\.", name, re.I):
        DIST.setdefault(key, []).append(i)
    else:
        LOC.setdefault(key, []).append(i)


for _alias, _canonical in PROVINCE_ALIASES.items():
    if norm(_canonical) in PROV and norm(_alias) not in PROV:
        PROV[norm(_alias)] = PROV[norm(_canonical)]


def prov_display(i):
    name = ROWS[i][1]
    m = re.match(r"\s*область\s+(.+)", name, re.I)
    if m:
        return m.group(1).strip() + " область"
    if re.search(r"област", name, re.I):
        return name.strip()
    return strip_markers(name)

def resolve_geo(segments):
    out = {"province": None, "city": None, "district": None, "geo_conf": 0.0}
    prov_id, city_text, ambiguous = None, None, False

    for seg in segments:
        text = pick_ru(seg)
        shown = strip_markers(text)
        key = norm(shown)
        if not key:
            continue
        is_dist = re.search(r"\b(район|р-н|ауданы)\b", text, re.I)
        is_prov = re.search(r"\b(область|облысы)\b|обл\.", text, re.I)

        if is_dist:
            out["district"] = shown
            if prov_id is None and key in DIST:
                tops = {top_of(i) for i in DIST[key]}
                if len(tops) == 1:
                    prov_id = tops.pop()
        elif is_prov and key in PROV:
            prov_id = PROV[key][0]
        elif key in LOC:
            city_text = shown
        elif key in PROV:
            prov_id = PROV[key][0]
        elif key in DIST:
            out["district"] = shown
        else:
            city_text = city_text or shown

    if city_text:
        out["city"] = city_text
        ids = LOC.get(norm(city_text), [])
        if prov_id:
            narrowed = [i for i in ids if top_of(i) == prov_id]
            ids = narrowed or ids
        city_ids = [i for i in ids if re.search(r"г\.а\.|^г\.|город|қаласы", ROWS[i][1], re.I)]
        village = any(re.search(r"\bсело\b|(?<!\w)с\.|посел|посёл|пос\.|ауылы", pick_ru(sg), re.I)
                      for sg in segments if norm(strip_markers(pick_ru(sg))) == norm(city_text))
        if not village:
            ids = city_ids or ids
        tops = {top_of(i) for i in ids}
        if len(tops) > 1:
            ambiguous = True
        if ids:
            district_ids = set()
            for item in ids:
                district = None
                parent = ROWS[item][0]
                while parent in ROWS:
                    if re.search(r'район|р\.а\.|ауданы', ROWS[parent][1], re.I):
                        district = parent
                        break
                    parent = ROWS[parent][0]
                district_ids.add(district)
            if not out['district'] and len(district_ids) == 1 and None not in district_ids:
                out['district'] = strip_markers(ROWS[next(iter(district_ids))][1])
            best = min(ids, key=depth_of)
            if depth_of(best) == 0:
                prov_id = best
            elif prov_id is None and not ambiguous:
                prov_id = top_of(best)

    if prov_id:
        out["province"] = prov_display(prov_id)

    score = 0.0
    score += 0.5 if out["province"] else 0
    score += 0.5 if out["city"] else 0
    out["geo_conf"] = score - (0.3 if ambiguous else 0)

    return out

import re
import unicodedata

LOW = 'а-яёәіңғүұқөһ'
UP = 'А-ЯЁӘІҢҒҮҰҚӨҺ'
BN = '\x01'  # «б/н» (без номера) как один символ

_CHUNK = re.compile(
    rf'(?P<num>\d+(?:/\d+)?(?:-\d+)?)'
    rf'|(?P<bn>{BN})'
    rf'|(?P<caps>[{UP}]{{2,}}(?![{LOW}])|[A-Z]{{2,}}(?![a-z]))'
    rf'|(?P<cap>[{UP}][{LOW}]*(?:-[{UP}][{LOW}]*)*|[A-Z][a-z]*(?:-[A-Z][a-z]*)*)'
    rf'|(?P<low>[{LOW}]+|[a-z]+)'
    rf'|(?P<slash>/)'
    rf'|(?P<other>\S)'
)

FIELD = {}
for _name, _words, _suffix in (
    ('house', ('д', 'дом', 'дм', 'house'), False),
    ('house', ('үй', 'үйі', 'уй'), True),
    ('apartment', ('кв', 'квартира', 'кварт', 'офис', 'оф', 'пом', 'помещ', 'помещение', 'flat', 'kv', 'kb', 'apt'), False),
    ('apartment', ('пәтер', 'пэтер', 'pater'), True),
    ('entrance', ('под', 'подъезд', 'подьезд', 'подезд', 'пд'), False),
    ('entrance', ('кіреберіс', 'киреберис'), True),
    ('floor', ('эт', 'этаж', 'floor'), False),
    ('floor', ('қабат', 'кабат'), True),
    ('building', ('корп', 'корпус', 'блок', 'block'), False),
    ('structure', ('стр', 'строение'), False),
):
    for _w in _words:
        FIELD[_w] = (_name, _suffix)

TYPE = {
    'ул': 'ул.', 'улица': 'ул.', 'көше': 'ул.', 'көшесі': 'ул.', 'ul': 'ул.',
    'пр': 'пр.', 'пр-т': 'пр.', 'пр-кт': 'пр.', 'проспект': 'пр.', 'даңғылы': 'пр.', 'даңғыл': 'пр.',
    'мкр': 'мкр.', 'мкрн': 'мкр.', 'микрорайон': 'мкр.', 'mkr': 'мкр.',
    'пер': 'пер.', 'переулок': 'пер.', 'бул': 'бул.', 'бульвар': 'бул.',
}
GEO = {  # суффиксы геообъектов в казахском: «Қарасайауданы»
    'ауданы': 'district', 'аудан': 'district', 'район': 'district',
    'ауылы': 'locality', 'ауыл': 'locality',
    'облысы': 'province', 'обл': 'province', 'область': 'province',
}
CITY_TAIL = {'ш', 'қ', 'қала', 'қаласы', 'г'}
LIVING = {'жк', 'жм'}
LETTER_MARKERS = set('дкспо')  # одиночные строчные буквы, которые чаще маркеры, чем литера дома

_MARKERS_BY_LEN = sorted(set(FIELD) | set(TYPE) | set(GEO), key=len, reverse=True)
_PEEL = sorted([m for m in set(FIELD) | set(GEO) | {'көшесі', 'көше', 'даңғылы', 'даңғыл'} if len(m) >= 2],
               key=len, reverse=True)
_ROMAN = re.compile(r'[IVXLC]+', re.I)


def _segment_lower(word):
    """'дкв' -> ['д','кв']; если разобрать целиком нельзя — [word]."""
    out, pos = [], 0
    while pos < len(word):
        for marker in _MARKERS_BY_LEN:
            if word.startswith(marker, pos):
                out.append(marker)
                pos += len(marker)
                break
        else:
            return [word]
    return out


def _peel_suffixes(word):
    """'Вкв' -> ('В',['кв']), 'Елдаңғылы' -> ('Ел',['даңғылы']), 'Қарасайауданы' -> ('Қарасай',['ауданы'])."""
    tail = []
    while True:
        low = word.lower()
        for marker in _PEEL:
            if low.endswith(marker) and len(word) > len(marker):
                head = word[:-len(marker)]
                # одиночную заглавную букву оставляем (литера корпуса), иначе голова >= 2 букв
                if len(head) >= 2 or head.isupper():
                    word = head
                    tail.insert(0, marker)
                    break
        else:
            return word, tail


def _chunks(text):
    items = []
    for m in _CHUNK.finditer(text):
        kind, value = m.lastgroup, m.group()
        if kind == 'other':
            continue
        items.append([kind, value])
    return items


def _find_city(word, is_city):
    """Самый длинный префикс-город + допустимый «хвост» (маркер типа улицы, ш/қ, обл...)."""
    low_tails = set(TYPE) | set(GEO) | CITY_TAIL | LIVING

    def tail_ok(rest):
        rest = rest.lower()
        if rest == '' or rest in low_tails:
            return True
        # опечатка вида «Атыраулмкр»: один лишний символ перед известным маркером
        return any(rest.endswith(t) and len(rest) - len(t) == 1 for t in low_tails if len(t) >= 2)

    for k in range(len(word), 2, -1):
        if is_city(word[:k]) and tail_ok(word[k:]):
            return word[:k], word[k:].lower()
    return None, None


def explode(raw, is_city):
    """Возвращает каноническую строку «г. Город, ул. Улица, д. 1, кв. 2» либо None."""
    s = unicodedata.normalize('NFKC', raw or '')
    s = re.sub(r'[\u200b-\u200f\u2060\ufeff\u00ad]', '', s).strip()
    if len(s) < 12 or re.search(r'[\s,;]', s) or not re.search(r'\d|б\s*/\s*н', s, re.I):
        return None  # обычная строка — её разберёт основной конвейер

    s = re.sub(r'б\s*/\s*н', BN, s, flags=re.I)
    s = re.sub(r'№|No\.?', '', s)
    raw_items = _chunks(s)

    # 1) раскладываем куски: отщепляем маркеры, режем строчные цепочки
    items = []
    for kind, value in raw_items:
        if kind == 'cap':
            head, tail = _peel_suffixes(value)
            items.append(['cap', head])
            items.extend(['low', t] for t in tail)
        elif kind == 'low':
            parts = _segment_lower(value)
            if len(value) == 1 and value not in FIELD and value not in TYPE:
                items.append(['letter', value])
            else:
                items.extend(['low', p] for p in parts)
        else:
            items.append([kind, value])

    # 1б) «Байганинад3/2» -> Байганина + д; «Авангард3д22» (д дальше есть) не трогаем
    for k in range(len(items) - 1):
        kind, value = items[k]
        if (kind == 'cap' and items[k + 1][0] == 'num' and value[-1:] in 'дД' and len(value) >= 5
                and value[-2].lower() not in 'рнлзг'
                and not any(it[0] == 'low' and it[1] in ('д', 'дом') for it in items[k + 2:])):
            items[k][1] = value[:-1]
            items.insert(k + 1, ['low', 'д'])
            break

    # 2) приставка «г»/«г.» перед городом
    if items and items[0][0] == 'low' and items[0][1] in ('г', 'гор', 'город'):
        items.pop(0)
    elif items and items[0][0] == 'letter' and items[0][1] == 'г':
        items.pop(0)
    if not items or items[0][0] != 'cap':
        return None

    # 3) город (+ слитный маркер после него). Город мог быть отщеплён «по хвосту» — склеим заново.
    first = items[0][1]
    consumed = 1
    while True:
        city, rest = _find_city(first, is_city)
        if city:
            break
        # «Усть-Каменогорск»/двухсловные города: пробуем присоединить следующий cap
        if consumed < len(items) and items[consumed][0] == 'cap' and consumed < 3:
            first += items[consumed][1]
            consumed += 1
            continue
        return None
    items = items[consumed:]
    # опечатка «Атыраулмкр»: хвост = лишняя буква + маркер -> берём сам маркер
    if rest not in TYPE and rest not in GEO and rest not in CITY_TAIL:
        rest = next((t for t in sorted(TYPE, key=len, reverse=True) if rest.endswith(t)), rest)
    province = None
    if rest == '' and items and items[0][0] == 'low' and GEO.get(items[0][1]) == 'province':
        rest = items.pop(0)[1]
    if rest in GEO and GEO[rest] == 'province':
        province, city = city, None
    elif rest in TYPE:
        items.insert(0, ['low', rest])
    # остальные хвосты (ш, қ, ...) — просто признак города

    # 4) автомат полей
    out = {'house': None, 'apartment': None, 'entrance': None, 'floor': None,
           'building': None, 'structure': None}
    street_type, names, living, district, locality = None, [], None, None, None
    n = len(items)
    i = 0

    def take_value(j):
        """Значение поля после маркера: число(+литера), римская цифра, литера(+число). -> (value, next_j)."""
        if j >= n:
            return None, j
        kind, value = items[j]
        if kind == 'num':
            j += 1
            if j + 1 < n and items[j][0] == 'slash' and items[j + 1][0] in ('letter', 'cap') and len(items[j + 1][1]) == 1:
                value += '/' + items[j + 1][1]
                j += 2
                return value, j
            if j < n and items[j][0] in ('letter', 'cap') and len(items[j][1]) == 1 and items[j][1].lower() not in LETTER_MARKERS:
                value += items[j][1]
                j += 1
            return value, j
        if kind == 'caps' and _ROMAN.fullmatch(value):
            return value, j + 1
        if kind in ('cap', 'letter') and len(value) == 1:
            letter = value
            j += 1
            if j < n and items[j][0] == 'num':
                letter += items[j][1]
                j += 1
            return letter, j
        if kind == 'bn':
            return 'б/н', j + 1
        return None, j

    while i < n:
        kind, value = items[i]
        low = value.lower()

        if kind == 'num':
            nxt = items[i + 1] if i + 1 < n else None
            nxt2 = items[i + 2] if i + 2 < n else None
            # «144үй», «34пәтер» — число перед казахским маркером
            if nxt and nxt[0] == 'low' and nxt[1] in FIELD and FIELD[nxt[1]][1]:
                out[FIELD[nxt[1]][0]] = value
                i += 2
                continue
            # «12д» + «кв…»: д — суффикс дома, а не префикс следующего числа
            if nxt and nxt[0] == 'low' and nxt[1] in ('д', 'дом') and (nxt2 is None or nxt2[0] != 'num'):
                out['house'] = value
                i += 2
                continue
            val, j = take_value(i)
            # у микрорайона первое число — часть названия («Аксай 4»), дом идёт после «д»
            if street_type == 'мкр.' and names and not names[-1][-1:].isdigit() and out['house'] is None \
                    and j < n and items[j][0] == 'low' and items[j][1] in ('д', 'дом'):
                names[-1] += ' ' + val
                i = j
                continue
            if out['house'] is None:
                out['house'] = val
            i = j
            continue

        if kind == 'bn':
            out['house'] = 'б/н'
            i += 1
            continue

        if kind == 'low' and low in FIELD:
            field, _ = FIELD[low]
            val, j = take_value(i + 1)
            if val is not None and out[field] is None:
                out[field] = val
                i = j
                continue
            i += 1
            continue

        if kind == 'low' and low in TYPE:
            street_type = TYPE[low]
            i += 1
            continue

        if kind == 'low' and low in GEO:
            target = GEO[low]
            if names:
                value_ = ' '.join(names)
                names = []
                if target == 'district':
                    district = value_
                elif target == 'locality':
                    locality = value_
                elif target == 'province':
                    province = value_
            i += 1
            continue

        if kind == 'caps' and low in LIVING:
            j, area = i + 1, []
            while j < n and items[j][0] in ('cap', 'caps') and items[j][1].lower() not in LIVING:
                area.append(items[j][1])
                j += 1
            living = ' '.join(area) or None
            i = j
            continue

        if kind in ('cap', 'caps'):
            names.append(value)
            i += 1
            continue

        i += 1  # неопознанный мусор и одиночные «/»

    name = ' '.join(names) or None
    if not (name or living) and out['house'] is None:
        return None

    parts = []
    if province:
        parts.append(f'{province} облысы')
    if district:
        parts.append(f'{district} ауданы')
    if city:
        parts.append(f'г. {city}')
    if locality:
        parts.append(f'{locality} ауылы')
    if living:
        parts.append(f'ЖК {living}')
    if name:
        parts.append(f'{street_type} {name}' if street_type else name)
    for field, label in (('house', 'д.'), ('building', 'корп.'), ('structure', 'стр.'),
                         ('entrance', 'под.'), ('floor', 'эт.'), ('apartment', 'кв.')):
        if out[field]:
            parts.append(f'{label} {out[field]}')
    return ', '.join(parts)

def as_record(item: Any) -> dict:
    """Любой элемент превращаем в запись: строка = raw_address, мусор = пустая запись."""
    if isinstance(item, dict):
        return item
    if isinstance(item, str):
        return {"raw_address": item}
    return {}


def extract_records(payload: Any) -> list:
    """Принимает голый список, {"records": [...]} и одиночную запись."""
    if isinstance(payload, list):
        return payload
    if isinstance(payload, dict):
        for key in LIST_KEYS:
            if isinstance(payload.get(key), list):
                return payload[key]
        if "raw_address" in payload:
            return [payload]
    return []


def parse_one(item: Any) -> dict:
    rec = as_record(item)
    raw = rec.get("raw_address")
    if raw is not None and not isinstance(raw, str):
        raw = str(raw)
    out = {"id": rec.get("id"), "raw_address": raw}
    for f in FIELDS:
        out[f] = None
    out["confidence"] = 0.0
    try:
        parsed = parse_address(raw or "")
        for f in FIELDS:
            out[f] = parsed.get(f)
        out["confidence"] = parsed.get("confidence", 0.0)
        out['review'] = parsed.get('review', {'needs_review': True, 'issues': []})
    except Exception:
        log.exception("parse failed for id=%s", rec.get("id"))
    return out


@app.get("/health")

@app.post("/parse")
def parse_single(payload: Any = Body(...)):
    records = extract_records(payload)
    return parse_one(records[0] if records else payload)


@app.post("/parse/batch")
def parse_batch(payload: Any = Body(...)):
    return [parse_one(r) for r in extract_records(payload)]

def _typo_aliases():
    words={word for group in (*FIELD_WORDS.values(),*STREET_WORDS.values()) for word in group if len(word)>=5}

    for word in words:

def extract_postal(s):
    label=r'(?:почтовый\s*индекс|пошталық\s*индекс|пошта\s*индексі|индекс[і]?|postal\s*code|postcode|zip\s*code|zipcode|zip)'
    explicit = re.compile(rf'(?<!\w){label}\s*[:№#.-]?\s*((?=[A-Z0-9]*\d)[A-Z0-9]{{4,10}})(?!\w)',re.I)
    matches=list(explicit.finditer(s))
    if not matches:
        matches=list(re.finditer(rf'(?<!\w)({POSTAL})\s*{label}(?!\w)',s,re.I))
    if not matches:
        for match in re.finditer(rf'(?<!\w){POSTAL}(?!\w)',s,re.I):
            before=s[:match.start()]
            if re.search(r'(?:дом|дм|үй|квартира|пәтер|bld|appt)\s*[.№#]?\s*$',before,re.I):
                continue
            matches.append(match)
    if not matches:
        return None,s
    first=matches[0]
    value=first.group(1) if first.lastindex else first.group()
    for match in reversed(matches):s=s[:match.start()]+' '+s[match.end():]
    return value.upper(),s


def join_spaced_numbers(s):
    """После явного маркера одиночные цифры образуют один номер; запятая — граница."""
    words = {word for group in FIELD_WORDS.values() for word in group}
    words |= {alias for alias, target in TYPO_ALIASES.items() if target in words}
    markers = '|'.join(re.escape(word) for word in sorted(words, key=len, reverse=True))
    pattern = rf'(?<!\w)({markers})(\.?\s*)(\d(?:[ \t]+\d){{1,5}})(?!\w|[ \t]+\d)'
    return re.sub(pattern, lambda m: m[1]+m[2]+re.sub(r'\s+', '', m[3]), s, flags=re.I)


def normalize_compact(s):
    """Разделяет маркеры на границах чисел и в начале слова, не режет произвольные названия."""
    s=re.sub(r'(?<!\w)((?i:ул|ul|мкрн|мкр|mkr|пр-т|пр))(?=[А-ЯӘІҢҒҮҰҚӨҺ])',r'\1 ',s)
    s=re.sub(
        r'(?P<prefix>[^\W\d_]*?)(?P<number>\d+[а-яa-z]?)(?P<marker>дом|дм|үйі?|уй)(?P<tail>/\d+[а-яa-z]?)',
        lambda m: (m['prefix'] + ' ' if m['prefix'] else '') + m['marker'] + ' ' + m['number'] + m['tail'],
        s, flags=re.I,
    )
    field_words={word for words in FIELD_WORDS.values() for word in words if len(word)>=2}
    street_words={word for words in STREET_WORDS.values() for word in words if len(word)>=3}
    street_suffixes='|'.join(re.escape(word) for word in sorted(street_words,key=len,reverse=True))
    s=re.sub(rf'(?<!\w)({street_suffixes})(?=\d)',r'\1 ',s,flags=re.I)
    s=re.sub(rf'(?<=[^\W\d_])({street_suffixes})(?=\d)',r' \1 ',s,flags=re.I)
    markers='|'.join(re.escape(word) for word in sorted(field_words|street_words,key=len,reverse=True))
    fields='|'.join(re.escape(word) for word in sorted(field_words,key=len,reverse=True))
    glued_following = field_words | street_words | {'магазин','паркинг','парковка','тц','трц','банк','школа','аптека','qazpost','жанында','рядом','возле'}
    following='|'.join(re.escape(word) for word in sorted(glued_following,key=len,reverse=True))
    s=re.sub(rf'(?<!\w)({fields})(?=(?:{following}))',r'\1 ',s,flags=re.I)
    s=re.sub(rf'(?<!\w)({fields})\s*[№#:=]\s*(?=\d)',r'\1 ',s,flags=re.I)
    s=re.sub(rf'(?<!\w)({fields})\s*no\.?\s*(?=\d)',r'\1 ',s,flags=re.I)
    s=re.sub(rf'(?<!\w)({fields})(?=[a-zа-я]\d)',r'\1 ',s,flags=re.I)
    house_markers='|'.join(re.escape(word) for word in sorted(FIELD_WORDS['house'],key=len,reverse=True))
    apartment_markers='|'.join(re.escape(word) for word in sorted(FIELD_WORDS['apartment'],key=len,reverse=True))
    s=re.sub(rf'(?<!\w)({house_markers})({apartment_markers})(?=\d)',r'\1 \2 ',s,flags=re.I)
    s=re.sub(rf'(?<=\d)({house_markers})\s*({apartment_markers})(?=\s*\d)',r' \1 \2 ',s,flags=re.I)
    s=re.sub(rf'(?<!\w)({fields})\?\s*',r'\1 ',s,flags=re.I)
    s=re.sub(rf'(?<!\w)({fields})\s*[—–]\s*(?=\d)',r'\1 ',s,flags=re.I)
    s=re.sub(r'(?<!\w)(ул|ul|st)\s*\.(?=\s|$)',r'\1 ',s,flags=re.I)
    s=re.sub(r'(?<!по)(?<![Пп][Оо])(?<=[^\W\d_])д\.\s*(?=\d)',' д',s)   # не рвём «под. 2» на «по д2»
    s=re.sub(rf'(?<=\d)-(?=(?:{markers}))',' ',s,flags=re.I)
    s=re.sub(rf'(?<=\d)(?=(?:{markers}))',' ',s,flags=re.I)
    s=re.sub(r'(?<=\d)(?=[^\W\d_]{2,})',' ',s,flags=re.U)
    full_markers=sorted(set(STREET_WORDS)|{'мкр','мкрн','көшесі','көше','даңғылы','даңғыл','тұйық','тұйығы','шағынаудан','ықшамаудан'},key=len,reverse=True)
    full_markers+=sorted({alias for alias,target in TYPO_ALIASES.items() if target in street_words and len(target)>=5},key=len,reverse=True)
    known=field_words|street_words
    def separate_name(match):
        token=match.group()
        if token.lower() in known or token.lower() in TYPO_ALIASES:return token

    s=re.sub(r'(?<=\w)(?=қаласы\b|облысы\b|ауданы\b)',' ',s,flags=re.I)

def split_compact_chains(s):
    s=re.sub(r'(?<!\w)(?:(?:дом|дм|д)\s*)?(\d+[а-яa-z]?(?:/\d+[а-яa-z]?)?)[- ]?(к|k|с|c)(\d+)(?!\w)',

    words={word for words in FIELD_WORDS.values() for word in words if len(word)>=2}
    words|={alias for alias,target in TYPO_ALIASES.items() if target in words}
    markers='|'.join(re.escape(word) for word in sorted(words,key=len,reverse=True))
    number=r'\d+[а-яa-z]??(?:/\d+[а-яa-z]??)?'
    streets='|'.join(re.escape(word) for words in STREET_WORDS.values() for word in sorted(words,key=len,reverse=True) if len(word)>2)
    prefix=re.compile(rf'(?:{markers}){number}(?=(?:{markers})|(?:{streets})|$)',re.I)
    suffix=re.compile(rf'{number}(?:{markers})(?=\d|(?:{markers})|(?:{streets})|$)',re.I)
    def separate(match):
        token=match.group()
        position=0;parts=[]
        while position<len(token):
            pattern=suffix if token[position].isdigit() else prefix
            pair=pattern.match(token,position)
            if not pair:break
            parts.append(pair.group());position=pair.end()
        if not parts:return token
        if position<len(token):parts.append(token[position:])
        return ' '.join(parts)
    return re.sub(r'[^\s,;]+',separate,s)

"""Поиск района и типа улицы по справочнику OSM (reference/osm_streets.sqlite).
Если файла нет, всё работает как раньше: функции возвращают None."""
import os

from app.normalization import CITY_ALIASES

PATH = os.path.join(os.path.dirname(__file__), "..", "reference", "osm_streets.sqlite")
ALIAS = {"нур-султан": "астана", "өскемен": "усть-каменогорск", "қонаев": "конаев", "капчагай": "конаев"}
TYPE_RE = r"улица|ул|проспект|пр-т|просп|пр|переулок|пер|бульвар|бул|шоссе|тупик|микрорайон|мкр|квартал|кв-л|көшесі|көше|кошеси|даңғылы|дангылы|даңғыл|тұйық көшесі|street|st|avenue|ave|road|rd"

_db = None
_cities = {}
_compact_streets = {}


def _compact_name(value):
    return re.sub(r'[\W_]+', '', _norm(value).translate(str.maketrans('әіңғүұқөһ', 'аингуукох')))


def _street_name_keys(value):
    """Conservative spelling variants for Kazakh street names written in Russian case."""
    key = _compact_name(value)
    keys = {key} if key else set()
    if len(key) >= 4:
        if key.endswith("я"):
            keys.add(key[:-1] + "й")
            keys.add(key[:-1])
        for ending, canonical in (("ова", "ов"), ("ева", "ев"), ("ина", "ин"), ("ына", "ын")):
            if key.endswith(ending) and len(key) > len(ending) + 2:
                keys.add(key[:-len(ending)] + canonical)
    return keys


def _conn():
    global _db
    if _db is None and os.path.exists(PATH):
        _db = sqlite3.connect(PATH, check_same_thread=False)
        _cities.update({r[0].lower(): r[0] for r in _db.execute("SELECT DISTINCT city FROM street")})
    return _db


def _city_record(city):
    """Resolve Russian, Kazakh and common Latin city names to the OSM key."""
    if not city:
        return None
    key = _norm(city).removeprefix("город ").strip()
    key = ALIAS.get(key, CITY_ALIASES.get(key, key))
    return _cities.get(key)


def _norm(t):
    t = t.lower().replace("ё", "е")
    t = re.sub(r"[^\w\s-]", " ", t)
    t = re.sub(r"\s+", " ", t).strip()
    return re.sub(rf"^(?:{TYPE_RE})\s+|\s+(?:{TYPE_RE})$", "", t)


def _place_district(db, c, name):
    """Микрорайон / жилой массив: ищем в таблице мест, отбрасывая номер в конце ('аксай 3а' -> 'аксай')."""
    n = _norm(name)
    cands = [n]
    base = re.sub(r"[\s-]+\d+[а-яa-z]?$", "", n)
    if base != n:
        cands.append(base)
    for cand in cands:
        rows = db.execute("SELECT district FROM place WHERE city=? AND (name=? OR name LIKE ? OR name LIKE ?)",
                          (c, cand, cand + " %", cand + "-%")).fetchall()
        ds = {r[0] for r in rows if r[0]}
        if len(ds) == 1:
            return ds.pop()
    return None


def lookup_place_district(city, name):
    """Return a district only when the local place reference maps the name uniquely."""
    db = _conn()
    if db is None or not city or not name:
        return None
    c = _city_record(city)
    return _place_district(db, c, name) if c else None


def lookup_street_type_global(street_name):
    """Infer a street type without a city only when the national OSM majority is clear."""
    db = _conn()
    if db is None or not street_name:
        return None
    name = _norm(street_name)
    rows = db.execute(
        "SELECT type, SUM(n) FROM street "
        "WHERE name=? OR name LIKE ? GROUP BY type",
        (name, "% " + name),
    ).fetchall()
    totals = {kind: count for kind, count in rows if kind and count}
    total = sum(totals.values())
    if not total:
        return None
    kind, count = max(totals.items(), key=lambda item: item[1])
    return kind if count / total >= 0.5 else None


def lookup(city, street_name, street_type=None):
    """-> {'district': str|None, 'street_type': str|None} или None."""
    db = _conn()
    if db is None or not city or not street_name:
        return None
    c = _city_record(city)
    if not c:
        return None
    out = {"district": None, "street_type": None}
    rows = db.execute("SELECT type, district, n FROM street WHERE city=? AND name=?",
                      (c, _norm(street_name))).fetchall()

def clean(s: str) -> str:
    if not s:
        return ""
    s = unicodedata.normalize("NFKC", s)
    s = re.sub(r"[\r\n]+", ", ", s)
    s = re.sub(r"[\u200b-\u200f\u2060\ufeff\u00ad]", "", s)
    s = s.replace("\xa0", " ")
    s = re.sub(r"(?<!\w)(?:мкр|мкрн|мик|мк)\.?\s*-\s*(?:н|он)(?!\w)\.?", "мкр", s, flags=re.I)   # «мкр.-н Алмагуль»
    s = re.sub(r"\.{2,}|…", ", ", s)                                # «г. Актобе... мкр. 11»
    s = re.sub(r"(?<!\w)(даңғылы|көшесі)ның\s+жалғасы(?!\w)", r"\1", s, flags=re.I)  # «Абай даңғылының жалғасы» -> «Абай даңғылы»
    s = re.sub(r"[!]+", " ", s)                                  # «кв. 14!!!»
    s = re.sub(r"(?<![A-Za-z])No\.?\s*(?=\d)", " ", s)           # NFKC превращает «№» в «No»: «кв. № 18»
    s = re.sub(r"(?<=[^\W\d_]\.)\s*-\s*(?=\d)", " ", s)        # «д.-14», «кв.-88»
    s = re.sub(r"(?<=\d)\s*[«\"“„]\s*([^\W\d_])\s*[»\"”“]", r"\1", s)  # «д. 5 «б»» -> «5б»
    s = re.sub(r"(?<!\w)н\s*\.\s*п\s*\.?(?!\w)", "нп", s, flags=re.I)
    s = re.sub(r"(?<!\w)к-сі(?!\w)", "көшесі", s, flags=re.I)
    s = re.sub(r"[;|]+", ",", s)
    s = re.sub(r"\s*/{2,}\s*", ", ", s)
    s = re.sub(r"\s+/\s+", ", ", s)
    s = re.sub(r"[()\[\]]", ", ", s)
    s = re.sub(r"\s+\.", ".", s)
    s = re.sub(r"\.(?=\S)", ". ", s)
    s = re.sub(r"(?:\s*,){2,}", ",", s)
    s = re.sub(r"\s+", " ", s).strip(" ,")
    return s

def extract_keyword_fields(s: str):
    tokens = re.findall(r",|[^\s,]+", s)
    norm = [canonical_marker(t) for t in tokens]
    n = len(tokens)
    is_num = [bool(NUM_RE.match(t)) for t in norm]

    kws = [i for i, t in enumerate(norm) if t in KW
           and not (t == "под" and i + 1 < n and norm[i + 1] in LANDMARK_INFLECTIONS)]

    def free_nums(i, used):
        cand = []
        field = KW[norm[i]]
        def usable(j):
            return is_num[j] or (field == "building" and bool(re.fullmatch(r"[a-zа-я](?:\d+[a-zа-я]?)?(?:-\d+)?", tokens[j], re.I)))
        if i > 0 and usable(i - 1) and (i - 1) not in used:
            cand.append(i - 1)
        if i + 1 < n and usable(i + 1) and (i + 1) not in used:
            cand.append(i + 1)
        return cand

    suffix_votes = sum(1 for i in kws
                       if i > 0 and is_num[i - 1] and not (i + 1 < n and is_num[i + 1]))
    prefix_votes = sum(1 for i in kws
                       if i + 1 < n and is_num[i + 1] and not (i > 0 and is_num[i - 1]))
    suffix_style = suffix_votes > prefix_votes

    fields, used, done = {}, set(), set()

    changed = True
    while changed:
        changed = False
        for i in kws:
            if i in done or KW[norm[i]] in fields:
                continue
            cand = free_nums(i, used)
            if len(cand) == 1:
                j = cand[0]
                fields[KW[norm[i]]] = tokens[j].strip(".")
                used.update({i, j})
                done.add(i)
                changed = True
        if changed:
            continue
        for i in kws:
            if i in done or KW[norm[i]] in fields:
                continue
            cand = free_nums(i, used)
            if len(cand) == 2:
                if KW[norm[i]] == "apartment":
                    j = cand[1]
                else:
                    j = cand[0] if suffix_style else cand[1]
                fields[KW[norm[i]]] = tokens[j].strip(".")
                used.update({i, j})
                done.add(i)
                changed = True
                break

    rest = [t for k, t in enumerate(tokens) if k not in used]
    return fields, " ".join(rest)

def join_house_letter(s):
    def space_fix(m):
        return m.group(0) if m.group(2).lower() in _NO_JOIN else m.group(1) + m.group(2)

    def dash_fix(m):
        return m.group(0) if m.group(2).lower() in "йм" else m.group(1) + m.group(2)

    s = re.sub(r"(?<![\w/])(\d+)\s+([^\W\d_])(?![^\W\d_])", space_fix, s)
    s = re.sub(r"(?<![\w/])(\d+)-([^\W\d_])(?![^\W\d_])", dash_fix, s)
    return s


def join_digit_runs(s):
    def fix(m):
        digits = m.group(0).split()
        if len(digits) >= 3 or "0" in digits:
            return "".join(digits)
        return m.group(0)

    return re.sub(r"(?<![\w/.,-])\d(?:[ \t]+\d)+(?![\w/-])", fix, s)

def extract_landmark(s):
    tokens = list(re.finditer(r",|[^\s,]+", s))
    for token in tokens:
        if canonical_marker(token.group()) in {"рынок", "базар"}:
            start = token.start()
            landmark = " ".join(s[start:].replace(",", " ").split()).strip()
            return landmark, s[:start].rstrip(" ,")
    descriptive_entrance = re.compile(
        r"^(?:со\s+двора|с\s+двора|со\s+стороны|с\s+улицы|возле\b|рядом\b|"
        r"жанында\b|сол\b|оң\b|паркинг\w*\b)", re.I
    )
    for i, token in enumerate(tokens):
        token_marker = canonical_marker(token.group())
        is_entrance_marker = token_marker in {word for word, field in KW.items() if field == "entrance"}
        if not is_entrance_marker and token_marker not in {"вход", "въезд"}:
            continue
        remainder = s[token.end():].lstrip(" ,")
        if not descriptive_entrance.search(remainder):
            continue
        comma = s.find(",", token.end())
        end = comma if comma >= 0 else len(s)
        landmark = " ".join(s[token.end():end].split()).strip(" ,")
        rest = s[:token.start()] + (s[comma + 1:] if comma >= 0 else "")
        return landmark, rest
    for i, token in enumerate(tokens):
        if canonical_marker(token.group()) in LANDMARK_SET | LANDMARK_INFLECTIONS:
            start = token.start()
            if canonical_marker(token.group()) == "ориентир":
                following = next((item for item in tokens[i + 1:] if item.group() != ","), None)
                if following is None:
                    continue
                start = following.start()
            if i and canonical_marker(tokens[i - 1].group()) in {"под", "над", "за", "перед", "у", "около"}:
                if canonical_marker(token.group()) != "ориентир":
                    start = tokens[i - 1].start()
            if not s[:start].strip(" ,"):
                comma = s.find(",", start)
                if comma != -1:
                    return ' '.join(s[start:comma].split()), s[comma + 1:]
            return ' '.join(s[start:].replace(',', ' ').split()), s[:start]
    return None, s


def extract_intersection(s):
    """Move a secondary crossing street into landmark, preserving the primary street."""
    marker = re.compile(r"(?<!\w)(?:уг\.?|угол|пересечение(?:\s+улиц(?:ы)?)?)(?!\w)", re.I)
    match = marker.search(s)
    if not match:
        return None, s
    tail = s[match.start():]
    end = len(tail)
    comma = tail.find(',')
    if comma >= 0:
        end = min(end, comma)
    boundary = re.search(r"(?<!\w)(?:дом|дм|д|үй|квартира|кв|пәтер|корпус|корп|подъезд|подьезд|этаж|эт)\.?\s*(?:\d|б/н)", tail, re.I)
    if boundary:
        end = min(end, boundary.start())
    landmark = " ".join(tail[:end].split()).strip(" ,")
    rest = (s[:match.start()] + " " + tail[end:]).strip(" ,")
    return landmark or None, rest


def extract_highway_route(s):
    """Protect a hyphenated route name and its kilometre point from generic parsing."""
    route = r"[^\s,]+(?:-[^\s,]+)+"
    kilometer = r"(?:\d+\s*(?:км|километр(?:а|ов)?)|км\.?\s*\d+|\d+\s*-\s*й\s+километр)"
    road_type = r"трассасы|трасса|тракт|шоссе|автодорога|автотрасса"
    patterns = (
        re.compile(rf"(?<!\w)(?P<km_before>{kilometer})\s+(?P<type>{road_type})\s+(?P<route>{route})(?:\s*,?\s*(?P<km_after>{kilometer}))?", re.I),
        re.compile(rf"(?<!\w)(?P<type>{road_type})\s+(?P<route>{route})(?:\s*,?\s*(?P<km_after>{kilometer}))?", re.I),
        re.compile(rf"(?<!\w)(?P<route>{route})\s+(?P<type>{road_type})(?:\s*,?\s*(?P<km_after>{kilometer}))?", re.I),
    )
    for pattern in patterns:
        match = pattern.search(s)
        if not match:
            continue
        marker = match.group('type').lower()
        normalized_type = 'шоссе' if marker == 'шоссе' else ('тракт' if marker.startswith('трасс') else 'тракт')
        kilometer_value = match.groupdict().get('km_before') or match.groupdict().get('km_after')
        name = match.group('route')
        if kilometer_value:
            name = f"{name} {kilometer_value.strip()}"
        rest = (s[:match.start()] + ' ' + s[match.end():]).strip(' ,')
        return {'street_type': normalized_type, 'street_name': name}, rest
    return None, s


def split_compact_geography(s):
    compact = lambda value: re.sub(r'[\s.-]+', '', value.lower())
    regions = {}
    for key, ids in PROV.items():
        if any('област' in ROWS[item][1].lower() for item in ids):
            regions[compact(key)] = prov_display(ids[0])
    for alias, canonical in PROVINCE_ALIASES.items():
        if compact(canonical) in regions:
            regions[compact(alias)] = regions[compact(canonical)]
    prefixes = {}
    for name, display in regions.items():
        for size in range(5, len(name)):
            prefixes.setdefault(name[:size], set()).add(display)
    for prefix, displays in prefixes.items():
        if len(displays) == 1:
            regions.setdefault(prefix, next(iter(displays)))
    cities = {compact(key): key for key in CITY_KEYS}
    from app.normalization import CITY_ALIASES
    cities.update({compact(alias): alias for alias in CITY_ALIASES})

    def separate(match):
        original = match.group()
        word = compact(original)
        if '-' in original:
            route_parts = [compact(part) for part in original.split('-')]
            if (all(part in cities for part in route_parts)
                    or (original[:1].isupper() and len(route_parts) >= 2
                        and all(part[:1].isupper() for part in original.split('-'))
                        and route_parts[0] in cities)):
                return original
        city_markers = sorted({compact(word) for words in STREET_WORDS.values() for word in words}, key=len, reverse=True)
        for prefix in ("г", "g"):
            for city in sorted(cities, key=len, reverse=True):
                if not word.startswith(prefix + city):
                    continue
                tail = word[len(prefix) + len(city):]
                marker = next((value for value in city_markers
                               if len(value) >= 2 and tail.startswith(value) and len(tail) > len(value) + 1), None)
                if marker:
                    city_end = len(prefix) + len(city)
                    marker_end = city_end + len(marker)
                    return (' ' + CITY_ALIASES.get(city, cities[city]) + ' '
                            + original[city_end:marker_end] + ' ' + original[marker_end:] + ' ')
        marker = re.search(r'(область|обл|облысы|облыс|oblast)$', word)
        if marker:
            stem = word[:marker.start()]
            if stem in regions:
                return ' ' + regions[stem] + ' '
            for city in sorted(cities, key=len, reverse=True):
                if stem.startswith(city) and stem[len(city):] in regions:
                    return ' ' + cities[city] + ' , ' + regions[stem[len(city):]] + ' '
        for city in sorted(cities, key=len, reverse=True):
            tail = word[len(city):] if word.startswith(city) else ""
            suffixes = ("қаласы", "қала", "qalasy", "qala", "каласы", "қ", "q")
            suffix = next((value for value in suffixes if tail.startswith(value)), None)
            if suffix and len(tail) == len(suffix):
                return ' ' + CITY_ALIASES.get(city, cities[city]) + ' ' + original[len(city):] + ' '
            if suffix and len(tail) > len(suffix):
                end = len(city) + len(suffix)
                return (' ' + CITY_ALIASES.get(city, cities[city]) + ' '
                        + original[len(city):end] + ' ' + original[end:] + ' ')
            street_abbreviations = {word for group in STREET_WORDS.values() for word in group}
            if tail in street_abbreviations:
                return ' ' + CITY_ALIASES.get(city, cities[city]) + ' ' + original[len(city):] + ' '
            if len(tail) >= 4:
                return ' ' + CITY_ALIASES.get(city, cities[city]) + ' ' + original[len(city):] + ' '
        return original
    s = re.sub(r'(область|облысы|oblast)(?=(?:дом|дм|кв|үй|пәтер)\d)', r'\1 ', s, flags=re.I)
    return re.sub(r'[^\W\d_]+(?:[-.][^\W\d_]+)*', separate, s)

        for alias in (stem + 'ский', stem + 'инский'):
            catalog.setdefault(alias, [])
            catalog[alias].extend(item for item in ids if item not in catalog[alias])
    return catalog


def extract_inline_geo(s):
    tokens = list(re.finditer(r",|[^\s,]+", s))
    taken, out = set(), {}
    for i, token in enumerate(tokens):
        if token.group().lower() in {'казахстан', 'қазақстан', 'kazakhstan', 'kazakstan', 'kz', 'рк', 'қр'} and (i == 0 or tokens[i - 1].group() == ',') \
                and (i + 1 >= len(tokens) or tokens[i + 1].group() == ','):
            taken.add(i)

    def claim(start, end, key, value):
        taken.update(range(start, end))
        out[key] = value

    def phrase(start, end):
        return ' '.join(token.group() for token in tokens[start:end])
    for i, token in enumerate(tokens):
        marker = canonical_marker(token.group())
        if marker == "қ" and i > 0 and geo_norm(tokens[i - 1].group()) in CITY_KEYS:
            taken.add(i)
            continue
        if i in taken or marker not in {"область", "обл", "облысы", "облыс", "oblast", "район", "р-н", "ауданы", "аудан"}:
            continue
        province = marker in {"область", "обл", "облысы", "облыс", "oblast"}
        catalog = PROV if province else district_catalog()
        match = None
        previous = tokens[i - 1].group() if i else ","
        suffix_district = marker in {"ауданы", "аудан", "район", "р-н"} and (
            previous != ","
            and geo_norm(previous) not in CITY_KEYS
            and canonical_marker(previous) not in set(STREET_TYPES) | set(KW)
        )
        directions = (-1, 1) if province else ((-1,) if suffix_district else (1, -1))
        for direction in directions:
            for count in (3, 2, 1):
                start, end = (i - count, i) if direction == -1 else (i + 1, i + 1 + count)
                if start < 0 or end > len(tokens) or any(j in taken or tokens[j].group() == ',' for j in range(start, end)):
                    continue
                value = phrase(start, end)
                key = geo_norm(strip_markers(value))
                if province and value.lower() in PROVINCE_ALIASES and geo_norm(PROVINCE_ALIASES[value.lower()]) in catalog:
                    key = geo_norm(PROVINCE_ALIASES[value.lower()])
                if key in catalog:
                    if province and not any(re.search(r'област|обл\.|облыс', ROWS[item][1], re.I) for item in catalog[key]):
                        continue
                    shown = prov_display(catalog[key][0]) if province else strip_markers(value)
                    match = min(start, i), max(end, i + 1), shown
                    break
            if match:
                break
        if not match and not province:
            if suffix_district:
                for count in (2, 1):
                    start = i - count
                    if start < 0 or any(j in taken or tokens[j].group() == ',' for j in range(start, i)):
                        continue
                    if count > 1 and geo_norm(tokens[start].group()) in CITY_KEYS:
                        start += 1
                    value = phrase(start, i)
                    if geo_norm(value) in {"белгісіз", "неизвестный", "неизвестно", "unknown"}:
                        break
                    if all(canonical_marker(tokens[j].group()) not in GEO_MARKERS | set(STREET_TYPES) | set(KW)
                           and not NUM_RE.match(tokens[j].group()) for j in range(start, i)):
                        match = start, i + 1, value
                        break
            else:
                for j in (i + 1, i - 1):
                    if 0 <= j < len(tokens) and j not in taken:
                        value = tokens[j].group()
                        if geo_norm(value) in {"белгісіз", "неизвестный", "неизвестно", "unknown"}:
                            continue
                        if value != ',' and canonical_marker(value) not in GEO_MARKERS | set(STREET_TYPES) | set(KW) and not NUM_RE.match(value):
                            match = min(i, j), max(i, j) + 1, value
                            break
        if not match and province:
            order = (i - 1, i + 1) if marker in {'облысы', 'облыс', 'oblast'} else (i + 1, i - 1)
            for j in order:
                if 0 <= j < len(tokens) and j not in taken:
                    value = tokens[j].group()
                    if value != ',' and canonical_marker(value) not in GEO_MARKERS | set(STREET_TYPES) | set(KW) and not NUM_RE.match(value):
                        match = min(i, j), max(i, j) + 1, value + ' ' + token.group()
                        break
        if match:
            claim(match[0], match[1], 'province' if province else 'district', match[2])
    for i, token in enumerate(tokens):
        marker = canonical_marker(token.group())
        if marker == 'қ' and i > 0 and geo_norm(tokens[i - 1].group()) in CITY_KEYS:
            taken.add(i)
            continue
        if i in taken or marker not in {'город', 'г', 'қала', 'қаласы', 'ауыл', 'ауылы', 'кент', 'село', 'пос', 'поселок', 'посёлок', 'с', 'п', 'ст'}:
            continue
        if marker in {'с', 'п', 'ст'} and not (i == 0 or tokens[i - 1].group() == ','):
            continue
        if marker == 'г' and i and canonical_marker(tokens[i - 1].group()) in STREET_TYPES:
            continue
        if marker in {'с', 'п', 'ст'} and (
                out.get('city')                                                    # город уже найден («г. Караганда, СТ ...»)
                or (i + 1 < len(tokens) and tokens[i + 1].group()[:1] in '"«“')   # «СТ "Наука"» — товарищество, не станция
                or (marker == 'ст' and token.group().isupper())):
            continue
        direction = -1 if marker in {'қаласы', 'ауылы'} else 1
        for count in (2, 1):
            start, end = (i - count, i) if direction < 0 else (i + 1, i + 1 + count)
            if start < 0 or end > len(tokens) or any(j in taken or tokens[j].group() == ',' for j in range(start, end)):
                continue
            value = phrase(start, end)
            if '/' in value:
                continue
            if geo_norm(value) in CITY_KEYS or (count == 1 and canonical_marker(value) not in GEO_MARKERS | set(STREET_TYPES) | set(KW) and not NUM_RE.fullmatch(value)):
                claim(min(i, start), max(i + 1, end), 'city', value)
                if marker in {'ауыл', 'ауылы', 'кент', 'село', 'пос', 'поселок', 'посёлок', 'с', 'п', 'ст'}:
                    out['_locality_kind'] = 'village'
                break
    for i in range(len(tokens)):
        if i in taken:
            continue
        for count in (2, 1):
            end = i + count
            if end > len(tokens) or any(j in taken or tokens[j].group() == ',' for j in range(i, end)):
                continue
            value = phrase(i, end)
            known_city = any(re.search(r'^г\.|город|қаласы', ROWS[item][1], re.I) for item in LOC.get(geo_norm(value), []))
            initial_city = i == 0 and end < len(tokens) and tokens[end].group() == ',' and known_city
            if geo_norm(value) not in CITY_KEYS and not initial_city:
                continue
            before = canonical_marker(tokens[i - 1].group()) if i else ''
            after = canonical_marker(tokens[end].group()) if end < len(tokens) else ''
            if (before in STREET_TYPES and after not in {'қаласы', 'қала'}) or before in LANDMARK_SET | LANDMARK_INFLECTIONS or before in {'область', 'обл', 'облыс', 'облысы', 'аудан', 'район'}:
                continue
            start = i - 1 if before in {'город', 'г', 'қала'} else i
            last = end + 1 if after in {'қаласы', 'қала'} else end
            claim(start, last, 'city', value)
            break
    if not out.get('district'):
        districts = district_catalog()
        city_tops = {top_of(item) for item in LOC.get(geo_norm(out.get('city', '')), [])}
        for i, token in enumerate(tokens):
            if i in taken:
                continue
            key = geo_norm(token.group())
            ids = districts.get(key, [])
            before = canonical_marker(tokens[i - 1].group()) if i else ''
            after = canonical_marker(tokens[i + 1].group()) if i + 1 < len(tokens) else ''
            if before in STREET_TYPES or after in STREET_TYPES or before in LANDMARK_SET | LANDMARK_INFLECTIONS:
                continue
            if city_tops:
                ids = [item for item in ids if top_of(item) in city_tops]
            elif not key.endswith(('ский', 'цкий')):
                continue
            if len(set(ids)) == 1:
                claim(i, i + 1, 'district', strip_markers(ROWS[ids[0]][1]))
                break
    spans = [(tokens[i].start(), tokens[i].end()) for i in taken]
    for start, end in sorted(spans, reverse=True):
        s = s[:start] + ' ' + s[end:]
    return out, s


def has_geo_conflict(res):
    cities = LOC.get(geo_norm(res.get('city') or ''), [])
    kind = r'село|с\.|пос|ауыл|аул|станц|ст\.' if res.get('_locality_kind') == 'village' else r'г\.а\.|^г\.|город|қаласы'
    cities = [i for i in cities if re.search(kind, ROWS[i][1], re.I)]
    provinces = PROV.get(geo_norm(strip_markers(res.get('province') or '')), [])
    return bool(cities and provinces and not ({top_of(i) for i in cities} & set(provinces)))


def canonical_marker(token):
    token = token.strip(".").lower()
    return MARKER_ALIASES.get(token, token)


def extract_living_area(s):
    marker = re.search(r"(?<!\w)(?:ж\s*[./]?\s*[мк]\.?|жк|жилмассив|жилой\s+массив|жилой\s+комплекс|тұрғын\s+алап|тұрғын\s+үй\s+кешені)(?!\w)\s*", s, re.I)
    if not marker:
        return None, s
    tokens = list(re.finditer(r",|[^\s,]+", s[marker.end():]))
    name_tokens = []
    end = marker.end()
    for token in tokens:
        value = token.group()
        key = canonical_marker(value)
        if value == "," or key in STREET_TYPES or key in KW or key in GEO_MARKERS or key in LANDMARK_SET:
            break
        name_tokens.append(value)
        end = marker.end() + token.end()
    if not name_tokens:
        return None, s
    return " ".join(name_tokens), s[:marker.start()] + " " + s[end:]


def nt(t):
    return canonical_marker(t)


def split_segments(tokens):
    segs, cur = [], []
    for t in tokens:
        if t == ",":
            if cur:
                segs.append(cur)
                cur = []
        else:
            cur.append(t)
    if cur:
        segs.append(cur)
    return segs


def is_geo_segment(seg):
    text = " ".join(seg)

    if re.search(r"[^\W\d_]/[^\W\d_]", text):
        return True
    return nt(seg[0]) in GEO_MARKERS or nt(seg[-1]) in GEO_MARKERS


_NOT_CITY = re.compile(r"село|посел|посёл|район|област|с\.о|а\.о|\bс\.|\bп\.|\bа\.|\bст\.|ауылы|ауданы|облысы", re.I)
CITY_KEYS = {"алматы", "астана", "нур-султан", "шымкент", "караганда", "актобе", "актау",
             "атырау", "костанай", "павлодар", "семей", "уральск", "тараз", "кызылорда",
             "петропавловск", "усть-каменогорск", "талдыкорган", "кокшетау", "туркестан",
             "темиртау", "экибастуз", "рудный", "жезказган", "балхаш", "конаев", "капчагай"}
for _i, (_parent, _name) in ROWS.items():
    if depth_of(_i) <= 1 and (not _NOT_CITY.search(_name) or re.search(r"город|г\.а\.|қаласы", _name, re.I)):
        _k = geo_norm(strip_markers(pick_ru(_name)))
        if _k:
            CITY_KEYS.add(_k)


def is_known_city(text):
    from app.normalization import CITY_ALIASES
    return geo_norm(text) in CITY_KEYS or text.lower() in CITY_ALIASES


def is_city_text(text):
    t = text.strip(". ")
    if not t or NUM_RE.match(t.lower()):
        return False
    return geo_norm(t) in CITY_KEYS


def split_leading_city(seg):
    if len(seg) < 2 or nt(seg[0]) in STREET_TYPES or nt(seg[0]) in GEO_MARKERS:
        return None, seg
    for k in (2, 1):
        if len(seg) <= k:
            continue
        cand = " ".join(seg[:k])
        if NUM_RE.match(nt(cand)):
            continue
        if is_city_text(cand):
            return cand, seg[k:]
    return None, seg


def parse_street(s, house=None):
    out = {"street_type": None, "street_name": None, "house": house,
           "landmark": None, "living_area": None, "_geo": []}
    tokens = re.findall(r",|[^\s,]+", s)

    for i, t in enumerate(tokens):
        if nt(t) in LANDMARK_SET:
            out["landmark"] = " ".join(x for x in tokens[i:] if x != ",")
            tokens = tokens[:i]
            break

    rest = []
    for seg in split_segments(tokens):
        if is_geo_segment(seg):
            out["_geo"].append(" ".join(seg))
        else:
            rest.append(seg)
    if not rest:
        return out

    city_txt, rest[0] = split_leading_city(rest[0])
    if city_txt:
        out["_geo"].append(city_txt)
    rest = [r for r in rest if r]
    if not rest:
        return out

    found = None
    for si, seg in enumerate(rest):
        for ti, t in enumerate(seg):
            if nt(t) in STREET_TYPES:
                found = (si, ti)
                break
        if found:
            break

    if found:
        si, ti = found
        seg = rest[si]
        stype = STREET_TYPES[nt(seg[ti])]
        out["street_type"] = stype
        out["_geo"] += [" ".join(x) for x in rest[:si]]
        later = rest[si + 1:]
        city_after = [x for x in later if is_city_text(" ".join(x))]
        out["_geo"] += [" ".join(x) for x in city_after]
        later = [x for x in later if x not in city_after]
        if stype not in NUMBER_BEFORE_TYPE and not out["living_area"]:
            for part in later:
                for area_pos, area_token in enumerate(part):
                    if nt(area_token) not in NUMBER_BEFORE_TYPE:
                        continue
                    area_name = part[:area_pos] + [area_token] + part[area_pos + 1:]
                    out["living_area"] = " ".join(area_name)
                    break
                if out["living_area"]:
                    break

        if stype == "микрорайон" and ti >= 1 and not NUM_RE.match(nt(seg[ti - 1])):
            name = seg[:ti]
            tail = seg[ti + 1:]
            if out["house"] is not None:
                name += tail
                tail = []
        elif stype in NUMBER_BEFORE_TYPE and ti >= 1 and NUM_RE.match(nt(seg[ti - 1])):
            if stype == "микрорайон":
                name = seg[:ti]
                out["living_area"] = None
            else:
                name = [seg[ti - 1]]
                out["living_area"] = " ".join(seg[:ti - 1]) or None
            tail = seg[ti + 1:]
        else:
            name = seg[ti + 1:]
            tail = []
            for offset, token in enumerate(name):
                if nt(token) in STREET_TYPES and STREET_TYPES[nt(token)] not in NUMBER_BEFORE_TYPE:
                    name = name[:offset]
                    break
            suffix_number = (stype not in NUMBER_BEFORE_TYPE and len(name) == 1
                             and NUM_RE.match(nt(name[0])))
            if ti > 0 and (not name or nt(name[0]) in KW or suffix_number):
                if suffix_number and out["house"] is None:
                    out["house"] = name[0].strip(".")
                name = seg[:ti]
            if (len(name) == 1 and NUM_RE.match(nt(name[0])) and ti > 0
                    and stype not in NUMBER_BEFORE_TYPE and out["house"] is None):
                out["house"] = name[0].strip(".")
                name = seg[:ti]
            elif not name and ti > 0:
                name = seg[:ti]
        if out['house'] is not None and stype not in NUMBER_BEFORE_TYPE and len(name) >= 2 and NUM_RE.fullmatch(nt(name[-1])):
            name = name[:-1]

        if out["house"] is None:
            for part in [tail] + later:
                if any(nt(token) in NUMBER_BEFORE_TYPE for token in part):
                    continue
                nums = [t for t in part if NUM_RE.match(nt(t))]
                if nums:
                    out["house"] = nums[0].strip(".")
                    break
            else:
                if len(name) >= 2 and NUM_RE.match(nt(name[-1])):
                    out["house"] = name[-1]
                    name = name[:-1]
        out["street_name"] = " ".join(name) or None
    else:
        if (len(rest) >= 2 and out["house"] is None
                and len(rest[-1]) == 1 and NUM_RE.match(nt(rest[-1][0]))):
            out["house"] = rest[-1][0]
            rest = rest[:-1]
        seg = rest[-1]
        out["_geo"] += [" ".join(x) for x in rest[:-1]]
        if out["house"] is None and len(seg) >= 2 and NUM_RE.match(nt(seg[-1])):
            out["house"] = seg[-1]
            seg = seg[:-1]
        elif out["house"] is None and len(seg) == 1 and NUM_RE.match(nt(seg[0])):
            out["house"] = seg[0]
            seg = []
        out["street_name"] = " ".join(seg) or None

    return out


def parse_address(raw: str) -> dict:
    res = {
        "province": None, "city": None, "district": None, "living_area": None,
        "street_type": None, "street_name": None, "house": None,
        "apartment": None, "entrance": None, "floor": None, "landmark": None,
        "postal_code": None, "building": None, "structure": None,
        "confidence": 0.0,
    }
    note_floor = note_entrance = None
    kept_notes = []
    for note in PARENTHETICAL_NOTE.findall(raw or ""):
        note = " ".join(note.split())
        if not note:
            continue
        m = re.fullmatch(r"(\d+)\s*-?\s*(?:й|ой|ый|я|ші)?\s*(?:этаж|қабат)", note, re.I)
        if m and not note_floor:
            note_floor = m.group(1)
            continue
        m = re.fullmatch(r"(\d+)\s*-?\s*(?:й|ой|ый|я|ші)?\s*(?:подъезд|подьезд|кіреберіс)", note, re.I)
        if m and not note_entrance:
            note_entrance = m.group(1)
            continue
        if re.fullmatch(r"[ВЗСЮБШ][КС]?[ОА]|ВКО|ЗКО|СКО|ЮКО|БКО|ШҚО|ҚО", note):   # аббревиатура области — не ориентир
            continue
        kept_notes.append(re.sub(r"^(?:ориентир|орентир)\s*[:\-–]?\s*", "", note, flags=re.I))
    parenthetical_notes = "; ".join(kept_notes) or None
    without_notes = PARENTHETICAL_NOTE.sub(" ", raw or "")
    # слитные строки без пробелов: «гУральскпрТулбаева12кв44» -> «г. Уральск, пр. Тулбаева, д. 12, кв. 44»
    exploded = glued.explode(without_notes, is_known_city)
    if exploded:
        without_notes = exploded
    source_text = strip_post_boxes(strip_access_notes(clean(without_notes)))
    no_house_number = bool(re.search(r"(?<!\w)(?:дом|дм|д|үй)?\s*б\s*/\s*н(?!\w)", source_text, re.I))
    if no_house_number:
        source_text = re.sub(r"(?<!\w)(?:дом|дм|д|үй)?\s*б\s*/\s*н(?!\w)", " ", source_text, flags=re.I)
    highway_route, s = extract_highway_route(source_text)
    from app.normalization import join_spaced_numbers
    s = join_spaced_numbers(s)
    s = join_digit_runs(s)
    postal_code, s = extract_postal(s)
    s = split_compact_geography(s)
    s = split_compact_chains(s)
    s = normalize_compact(s)
    fused_street_suffix = None
    suffix_match = re.search(
        r"(?<!\w)(?:улица|ул|проспект|пр-т|пр|street|st)\s+(.+?)(\d+)\s*(?=(?:дом|дм|д|үй)\s*\d)",
        s, re.I,
    )
    if suffix_match:
        fused_street_suffix = f"{suffix_match.group(1).strip()} {suffix_match.group(2)}"
    glued_fields, s = extract_glued_fields(s)
    s = normalize_compact(s)
    s = split_glued(s)
    s = join_house_letter(s)
    s = split_compact_geography(s)
    living_area, s = extract_living_area(s)
    fields, s = extract_keyword_fields(s)
    fields.update(glued_fields)
    for k, v in fields.items():
        res[k] = v

    explicit_geo, s = extract_inline_geo(s)
    intersection, s = extract_intersection(s)
    landmark, s = extract_landmark(s)

    st = parse_street(s, res["house"])
    if highway_route:
        st['street_type'] = highway_route['street_type']
        st['street_name'] = highway_route['street_name']
    if fused_street_suffix and st.get("street_type") and st.get("street_name"):
        st["street_name"] = fused_street_suffix
    pair = re.fullmatch(r"(.+?)\s+(\d+)-(\d+)", st.get("street_name") or "")
    if (pair and not res.get("house") and not st.get("house") and not res.get("apartment")):
        st["street_name"] = pair.group(1).strip()
        st["house"] = pair.group(2)
        res["apartment"] = pair.group(3)
    house_pair = re.fullmatch(r"(\d+)-(\d+)", str(st.get("house") or res.get("house") or ""))
    if house_pair and not res.get("apartment"):
        st["house"] = house_pair.group(1)
        res["house"] = house_pair.group(1)
        res["apartment"] = house_pair.group(2)
    if (st["house"] is None and st["street_name"] and res.get("entrance")
            and "/" in str(res["entrance"])):
        st["house"] = res["entrance"]
        res["entrance"] = None
    for k in ("street_type", "street_name", "house", "landmark", "living_area"):
        res[k] = st[k]
    if no_house_number:
        res["house"] = "б/н"
    if res["street_name"]:
        street_name = re.sub(r"\s+(?:д|дм|дом)\.?\s*$", "", res["street_name"], flags=re.I)
        street_name = re.sub(r"^[\s.,;:№#]+|[\s.,;:№#]+$", "", street_name)
        street_name = re.sub(r"\s+", " ", street_name).strip()
        numeric_quarter = (res.get("street_type") in NUMBER_BEFORE_TYPE
                           and bool(NUM_RE.fullmatch(street_name)))
        if not re.search(r"[^\W\d_]", street_name, re.UNICODE) and not numeric_quarter:
            street_name = None
        res["street_name"] = street_name or None
    if living_area:
        res["living_area"] = living_area
    if res.get("street_name"):
        name = str(res["street_name"]).strip()
        if res.get("street_type") not in NUMBER_BEFORE_TYPE and not highway_route:
            house_tail = re.fullmatch(r"(.+\D)\s+(\d+[a-zа-я]?(?:/\d+[a-zа-я]?)?)", name, re.I)
            if house_tail and not res.get("house"):
                res["street_name"] = house_tail.group(1).strip()
                res["house"] = house_tail.group(2)
            elif house_tail and res.get("house") == house_tail.group(2):
                res["street_name"] = house_tail.group(1).strip()
        res["street_name"] = re.sub(r"\s+(?:код|дом|дм|д|үй)\s*$", "", str(res["street_name"]), flags=re.I).strip()

            res["street_name"] = None

        res["landmark"] = landmark
    if intersection:
        res["landmark"] = "; ".join(value for value in (intersection, res.get("landmark")) if value)
    if parenthetical_notes:
        res["landmark"] = "; ".join(value for value in (res.get("landmark"), parenthetical_notes) if value)
    res["floor"] = res.get("floor") or note_floor
    res["entrance"] = res.get("entrance") or note_entrance

        geo_value = res.get(geo_field)

                res["city"] = candidate

        from app.geo import DIST, ROWS, strip_markers, top_of
        district_key = geo_norm(strip_markers(res["district"]))
        district_ids = DIST.get(district_key, [])
        city_ids = LOC.get(geo_norm(res.get("city") or ""), [])
        city_tops = {top_of(item) for item in city_ids}
        matching = [item for item in district_ids if not city_tops or top_of(item) in city_tops]

            res["district"] = strip_markers(ROWS[matching[0]][1])
        else:
            district = str(res["district"]).strip()
            res["district"] = district[:1].upper() + district[1:]


    if not res.get("district") and res["city"] and res.get("living_area") and res.get("_locality_kind") != 'village':
        area_district = osm.lookup_place_district(geo_norm(res["city"]), res["living_area"])
        if area_district:
            res["district"] = area_district

    if res["city"] and res["street_name"] and res.get('_locality_kind') != 'village':
        info = osm.lookup(geo_norm(res["city"]), res["street_name"], res["street_type"])
        street_match = 'matched' if info else 'not_confirmed'
        if info:
            if not res["district"] and info.get("district"):
                res["district"] = info["district"]
            if not res["street_type"] and info.get("street_type"):
                res["street_type"] = info["street_type"]
    elif not res["city"] and res["street_name"] and not res["street_type"]:
        inferred_type = osm.lookup_street_type_global(res["street_name"])
        if inferred_type:
            res["street_type"] = inferred_type
            street_match = 'global_type_inferred'

    res['postal_code'] = postal_code
    from app.review import build_review
    res['review'] = build_review(raw, res, street_match=street_match)
    issue_codes = {issue['code'] for issue in res['review']['issues']}
    route_kilometer = bool(res.get('street_type') in {'тракт','шоссе'} and res.get('street_name')
                           and re.search(r'(?:\d+\s*км|км\.?\s*\d+|\d+\s*-\s*й\s+километр)',res['street_name'],re.I))

        confidence = 0.35
    elif 'unassigned_numbers' in issue_codes or has_geo_conflict(res):
        confidence = 0.55
    elif not (res.get('street_name') or res.get('living_area')):
        confidence = 0.65
    elif street_match == 'not_confirmed':
        confidence = 0.78
    elif street_match == 'matched':
        confidence = 0.95
    else:
        confidence = 0.90
    city_key = geo_norm(res.get('city') or '')
    if res.get('city') and res.get('_locality_kind') != 'village' and city_key not in CITY_KEYS and not LOC.get(city_key):
        confidence = min(confidence, 0.6)           # город неизвестен справочнику
    street = res.get('street_name') or ''
    if res.get('street_name') and (len(re.sub(r'\W', '', street)) <= 1 or not re.search(r'[^\W\d_]', street)) \
            and res.get('street_type') not in NUMBER_BEFORE_TYPE:
        confidence = min(confidence, 0.5)           # название из одной буквы / без букв — почти наверняка обрезок
    if re.fullmatch(r'[IVXLC]+', str(res.get('house') or ''), re.I):
        confidence = min(confidence, 0.6)           # римская цифра вместо номера дома
    if not res.get('province') and not res.get('city'):
        confidence = min(confidence, 0.3)
    res['confidence'] = confidence
    return res

    critical_codes = {'missing_house', 'unassigned_numbers', 'geo_conflict', 'unbound_house'}
    needs_review = any(issue['code'] in critical_codes for issue in issues)
    return {'needs_review':needs_review,'issues':issues,'method':'rules-v2','confidence_kind':'heuristic','street_match':street_match,'locality_kind':result.get('_locality_kind','unspecified')}

BASE_CITIES = set(CITIES)   # города, для которых районы берём из OSM

HIGHWAYS = {"residential", "primary", "secondary", "tertiary", "unclassified",
            "living_street", "pedestrian", "trunk", "service", "road"}

TYPE_WORDS = [
    ("улица", r"улица|ул|көшесі|кошесі"),
    ("проспект", r"проспект|пр-т|просп|пр|даңғылы"),
    ("переулок", r"переулок|пер|тұйық көшесі"),
    ("бульвар", r"бульвар|бул|бульвары"),
    ("шоссе", r"шоссе|тас жолы"),
    ("тупик", r"тупик|тұйығы"),
    ("микрорайон", r"микрорайон|мкр|шағын ауданы"),
    ("квартал", r"квартал|кв-л"),
]

DISTRICT_FIX = {"Әулиеата": "Аулиеата", "Жібек жолы": "Жибек Жолы"}
DISTRICT_RE = re.compile(
    r"(?:район|р-н|районы|аудан(?:ы|ы)?|district|район города|қалалық аудан)", re.I
)

NEAR = 0.0012
DEFAULT_MARGIN = 0.10
RELAXED_MARGIN = {"Караганда": 0.5, "Актобе": 0.5}


def norm(t):
    t = t.lower().replace("ё", "е")
    t = re.sub(r"[^\w\s-]", " ", t)
    return re.sub(r"\s+", " ", t).strip()


def split_type(name):
    n = norm(name)
    for stype, words in TYPE_WORDS:
        m = re.match(rf"^(?:{words})\s+(.+)$", n) or re.match(rf"^(.+?)\s+(?:{words})$", n)
        if m:
            return stype, m.group(1).strip()
    return None, n


def city_of(lat, lon):
    for c, (s, w, n, e) in CITIES.items():
        if s <= lat <= n and w <= lon <= e:
            return c
    return None


# --- Автоматическое расширение списка городов (запуск с флагом --towns) ---
# Для каждого узла place=city/town из дампа OSM, который не попал в CITIES выше,
# добавляется прямоугольник вокруг центра. Одноимённые населённые пункты
# пропускаются: их улицы нельзя надёжно разделить по названию города.
TOWN_HALF_LAT = 0.07   # примерно 8 км на север и юг
TOWN_HALF_LON = 0.10   # примерно 7 км на восток и запад


def is_good_town_name(nm):
    """Только кириллица, дефис, пробел, точка; без цифр и иероглифов; без слов вроде «село имени»."""
    n = nm.strip().lower()
    if not re.fullmatch(r"[а-яёәіңғүұқөһ\- .]+", n):
        return False
    return not re.match(r"^(?:село|поселок|посёлок|имени|аул)\b", n)


def collect_towns(pbf):
    import osmium
    try:
        nodes_only = osmium.osm.osm_entity_bits.NODE
    except AttributeError:
        nodes_only = osmium.osm.NODE
    fp = osmium.FileProcessor(pbf, nodes_only).with_filter(osmium.filter.KeyFilter("place"))
    found = {}
    for o in fp:
        if not o.is_node():
            continue
        tags = o.tags
        if tags.get("place") not in ("city", "town"):
            continue
        nm = ru_name(tags)
        if not nm or "/" in nm or not is_good_town_name(nm):
            continue
        found.setdefault(norm(nm), []).append((nm.strip(), o.location.lat, o.location.lon))
    return found


def add_towns(found):
    added, duplicates, inside = [], [], []
    for key in sorted(found):
        items = found[key]
        if len(items) > 1:
            duplicates.append(items[0][0])
            continue
        nm, lat, lon = items[0]
        if nm in CITIES:
            continue
        if city_of(lat, lon):
            inside.append(nm)
            continue
        CITIES[nm] = (lat - TOWN_HALF_LAT, lon - TOWN_HALF_LON,
                      lat + TOWN_HALF_LAT, lon + TOWN_HALF_LON)
        added.append(nm)
    return added, duplicates, inside


def clean_district(name):
    n = DISTRICT_RE.sub(" ", name)
    n = re.sub(r"\s+", " ", n).strip(" .,-")
    return DISTRICT_FIX.get(n, n)


def ru_name(tags):
    return tags.get("name:ru") or tags.get("name")


def sample_points(nodes):
    total = len(nodes)
    if total == 0:
        return []
    idxs = sorted({0, total // 4, total // 2, (3 * total) // 4, total - 1})
    pts = []
    for i in idxs:
        try:
            nd = nodes[i]
            pts.append((nd.lat, nd.lon))
        except Exception:
            continue
    return pts


def main(pbf, out, towns=False):
    import osmium
    import shapely.wkb as wkb
    from shapely.geometry import Point
    from shapely.prepared import prep

    if towns:
        added, duplicates, inside = add_towns(collect_towns(pbf))
        print(f"Добавлено городов и посёлков: {len(added)}")
        print(f"Пропущено из-за одинаковых названий: {len(duplicates)}: {duplicates[:20]}")
        print(f"Пропущено, так как уже внутри другого города: {len(inside)}")

    streets = []
    places = []
    districts = []
    rejected = []
    wkbf = osmium.geom.WKBFactory()

    fp = (osmium.FileProcessor(pbf)
          .with_locations()
          .with_areas(osmium.filter.KeyFilter("boundary")))
    cnt = 0
    for o in fp:
        cnt += 1
        if cnt % 2_000_000 == 0:
            print(f"  ...обработано {cnt:,} объектов", flush=True)
        tags = o.tags
        if o.is_way() and "highway" in tags and tags["highway"] in HIGHWAYS:
            nm = ru_name(tags)
            if not nm:
                continue
            pts = sample_points(o.nodes)
            if not pts:
                continue
            city = city_of(*pts[len(pts) // 2])
            if not city:
                continue
            stype, nn = split_type(nm)
            if nn:
                streets.append((city, stype, nn, pts))
        elif o.is_node() and tags.get("place") in ("suburb", "neighbourhood", "quarter"):
            nm = ru_name(tags)
            if nm:
                city = city_of(o.location.lat, o.location.lon)
                if city:
                    places.append((city, o.location.lat, o.location.lon, nm))
        elif o.is_area() and tags.get("boundary") == "administrative":
            nm = ru_name(tags) or ""
            if not DISTRICT_RE.search(nm):
                continue
            try:
                poly = wkb.loads(wkbf.create_multipolygon(o), hex=True)
                rp = poly.representative_point()
            except Exception:
                continue
            city = city_of(rp.y, rp.x)
            if not city or city not in BASE_CITIES:
                continue
            s_, w_, n_, e_ = CITIES[city]
            minx, miny, maxx, maxy = poly.bounds
            m = RELAXED_MARGIN.get(city, DEFAULT_MARGIN)
            if miny < s_ - m or minx < w_ - m or maxy > n_ + m or maxx > e_ + m:
                rejected.append((city, nm, tags.get("admin_level", "?"),
                                 round(miny, 3), round(minx, 3), round(maxy, 3), round(maxx, 3)))
                continue
            districts.append((city, clean_district(nm), prep(poly), poly))

    print(f"Улиц (участков): {len(streets):,}; мест: {len(places):,}; районов: {len(districts)}")
    by_city = Counter(c for c, *_ in districts)
    print("Районов по городам:", dict(by_city))
    print("Улиц по городам:", dict(Counter(c for c, *_ in streets)))
    print("Города без районов:", [c for c in CITIES if c not in by_city])
    if rejected:
        print("\nОТКЛОНЁННЫЕ районы (город | название | admin_level | юг запад север восток):")
        for r in rejected[:60]:
            print("  ", r)
        print("Если среди них есть настоящие районы города, пришли этот список целиком.")
    if not districts:
        print("ВНИМАНИЕ: районы не найдены. Пришлите мне этот вывод целиком.")

    def district_at(city, lat, lon):
        p = Point(lon, lat)
        own = [(name, pp, poly) for c, name, pp, poly in districts if c == city]
        for name, pp, poly in own:
            if pp.contains(p):
                return name
        best, best_d = None, NEAR
        for name, pp, poly in own:
            d = poly.distance(p)
            if d < best_d:
                best, best_d = name, d
        return best

    agg = Counter()
    for city, stype, nn, pts in streets:
        for lat, lon in pts:
            agg[(city, nn, stype, district_at(city, lat, lon))] += 1

    if os.path.exists(out):
        os.remove(out)
    db = sqlite3.connect(out)
    db.execute("CREATE TABLE street(city TEXT, name TEXT, type TEXT, district TEXT, n INT)")
    db.executemany("INSERT INTO street VALUES (?,?,?,?,?)",
                   [(c, nn, st, d, k) for (c, nn, st, d), k in agg.items()])
    db.execute("CREATE INDEX ix_street ON street(city, name)")
    db.execute("CREATE TABLE place(city TEXT, name TEXT, district TEXT)")
    db.executemany("INSERT INTO place VALUES (?,?,?)",
                   [(c, norm(nm), district_at(c, la, lo)) for c, la, lo, nm in places])
    db.commit()

    print("\nДоля точек улиц с районом по городам с районами:")
    for city in by_city:
        total, with_d = db.execute(
            "SELECT SUM(n), SUM(CASE WHEN district IS NOT NULL THEN n ELSE 0 END) FROM street WHERE city=?",
            (city,)).fetchone()
        print(f"  {city}: {with_d}/{total} = {round(100 * with_d / total)}%")
    db.close()
    print("Готово:", out)


if __name__ == "__main__":
    if len(sys.argv) < 2:
        sys.exit("Использование: python scripts/build_osm.py путь/к/kazakhstan-latest.osm.pbf [--towns]")
    out = os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "reference", "osm_streets.sqlite")
    main(sys.argv[1], out, towns="--towns" in sys.argv[2:])
